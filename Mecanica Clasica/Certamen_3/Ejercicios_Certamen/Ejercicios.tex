\section{Problema 1}
The transformation of coordinates and momenta has a form
\begin{align}
	P &= \alpha\frac{p^2 q}{\sqrt{1 + p^{2}q^{2}}}, & Q &= \beta \frac{\sqrt{1 + p^2 q^2}}{p}, & \alpha, \beta &= \mathrm{const}
\end{align}
\begin{enumerate}
	\item[a.] Find the values of constants $\alpha, \beta$ for which the transformation is canonical. For these values of $\alpha, \beta$, find explicitly the generating function $F$ which corresponds to the transformation.
	\resp Podemos calcular el corchete de Poisson entre $[P,Q]_{p,q}$, es cual deberá ser cero. Usando \texttt{Mathematica} tenemos
	\begin{equation}
		[P,Q]_{p,q} = \alpha\beta
	\end{equation}
	Por lo tanto, para que la transformación sea canónica deberá cumplirse que
	\begin{equation}
		\beta = \frac{1}{\alpha}
	\end{equation}
	
	Para la ecuación generatriz, buscaremos una ecuación del tipo $F_1(q,Q,t)$ (porque las revisé todas y esta se ve la más simple de resolver :p). Escribiendo $p = p(q,Q), ~ P = P(q,Q)$ con \texttt{Mathematica}
	\begin{equation}
		\pd{F_1}{q} = p = \frac{1}{\sqrt{\alpha ^2 Q^2 - q^2}}, \qquad -\pd{F_1}{Q} = P = \frac{q}{Q \sqrt{\alpha ^2 Q^2 - q^2}}
	\end{equation}
	Usando \texttt{Mathematica} para integrar $p$ por $q$ y $P$ por $-Q$ llegamos al mismo resultado
	\begin{equation}
		F_1(q,Q) = \arctan\left[\frac{q}{\sqrt{\alpha ^2 Q^2-q^2}}\right]
	\end{equation}
	
	\qed
	
	\item[b.] Apply the transformation found in the previous item to the hamiltonian of harmonic oscillator $H (p, q) = \frac{p^2}{2m} + \frac{kq^2}{2}$ and find the hamiltonian $\tilde{H} (P, Q)$ in terms of the new variables $(P, Q)$. Write out the canonical equations in terms of these new variables.
	\resp Usando la función generatriz, podemos transformar el hamiltoniano utilizando
	\begin{equation}
		\tilde{H} = H + \pd{F_1}{t}
	\end{equation}
	Pero $\ltd{F}{t} = 0$. Primero, para escribir el hamiltoniano original en función de $P,Q$, necesitamos $p(P,Q), ~ q(P,Q)$. Usando \texttt{Mathematica}
	\begin{equation}
		q = \frac{\alpha  P Q^2}{\sqrt{P^2 Q^2+1}}, \qquad p = \frac{\sqrt{P^2 Q^2+1}}{\alpha  Q}
	\end{equation}
	por lo que el hamiltoniano
	\begin{equation}
		\tilde{H} = \frac{p^2}{2m} + \frac{kq^2}{2} = \frac{1}{2 m \alpha^2} \frac{P^2 Q^2 + 1}{ Q^2} + \frac{k\alpha^2}{2} \frac{P^2 Q^4}{P^2 Q^2 + 1}
	\end{equation}
	Y las ecuaciones canónicas, con \texttt{Mathematica}
	\begin{equation}
		\dot{P} = - \pd{\tilde{H}}{Q} = \frac{1}{\alpha ^2 m Q^3}-\frac{\alpha ^2 k P^2 Q^3 \left(P^2 Q^2+2\right)}{\left(P^2 Q^2+1\right)^2}, \qquad \dot{Q} = \pd{\tilde{H}}{P} = \frac{\alpha^2 k P Q^4}{ \left(P^2 Q^2+1\right)^2} + \frac{P}{\alpha ^2 m}
	\end{equation}
	\qed	
\end{enumerate}

\section{Problema 2}
A nonrelativistic charged particle of mass $m$ is constrained to move in a horizontal plane, in the field of the harmonic oscillator $U = kr^2/2$ and the constant magnetic field $\n{B}$ perpendicular to the plane. The vector potential $\n{A}$ may be chosen as
\begin{equation}
	\n{A} = \frac{1}{2} \n{B}\times\rv
\end{equation}
Analyze this system in polar coordinates:
\begin{enumerate}
	\item[a.] Write out the lagrangian, the generalized momenta and construct the hamiltonian.
	\resp Asumiendo que la partícula tiene carga $q$, el potencial magnético:
	\begin{equation*}
		U_A = - q \dot{\rv}\cdot\n{A} = - q \dot{\rv}\cdot\frac{1}{2} (\n{B}\times\rv)
	\end{equation*}
	Como $\n{B}$ es perpendicular a $\rv$, $\n{B}\times\rv = B r \phat$
	\begin{equation}
		U_A = - \frac{1}{2} q B r \dot{\rv}\cdot\phat = - \frac{q B}{2} r^2 \dot{\varphi}
	\end{equation}
	Entonces el lagrangiano
	\begin{equation}
		L = \frac{m}{2} \left(\dot{r}^2 + r^2 \dot{\varphi}^2\right) - \frac{ k}{2} r^2 + \frac{q B}{2} r^2 \dot{\varphi}
	\end{equation}
	Y los momentos generalizados
	\begin{equation}
		p_r = \pd{L}{\dot{r}} = m\dot{r}, \qquad p_\varphi = \pd{L}{\dot{\phi}} = m r^2 \dot{\varphi} + \frac{q B}{2} r^2
	\end{equation}
	Ya que $\varphi$ no aparece, es cíclica y $p_{\varphi}$ se conserva. El hamiltoniano entonces
	\begin{align*}
		H &= m \dot{r}^2 + m r^2 \dot{\varphi}^2 + \frac{q B }{2} r^2 \dot{\varphi} - \frac{m}{2} \left(\dot{r}^2 + r^2 \dot{\varphi}^2\right) + \frac{k}{2} r^2 - \frac{q B }{2} r^2 \dot{\varphi}\\
		&= \frac{m }{2} \dot{r}^2 + \frac{m }{2} r^2 \dot{\varphi}^2 + \frac{k}{2} r^2
	\end{align*}
	Usando los momentos
	\begin{equation}
		H = \frac{1}{2 m} p_r^2 + \frac{1}{2 m} \left(\frac{p_\varphi}{r} - \frac{ q B}{2} r\right)^2 + \frac{k}{2} r^2
	\end{equation}
	
	\qed
	
	\item[b.] Write out the canonical equations of motion.
	\resp Las ecuaciones canónicas
	\begin{align}
		\dot{p}_r &= - \pd{H}{r} = - \frac{1}{m} \left(\frac{p_\varphi}{r} - \frac{ q B}{2} r\right) \left(- \frac{p_\varphi}{r^2} - \frac{ q B}{2}\right) - k r = \frac{1}{m} \frac{p_\varphi^2}{r^3} - \left[\frac{q^2 B^2 }{4 m} + k\right] r\\
		\dot{p}_\varphi &= - \pd{H}{\varphi} = 0 \\
		\dot{r} &= \pd{H}{p_r} = \frac{1}{m} p_r\\
		\dot{\varphi} &= \pd{H}{p_\varphi} = \frac{1}{m} \left(\frac{p_\varphi}{r} - \frac{ q B}{2} r\right) \cdot \frac{1}{r} = \frac{1}{m} \frac{p_\varphi}{r^2} - \frac{ q B}{2 m}
	\end{align}
	
	\qed
	
	\item[c.] Write out the equation of Hamilton-Jacobi and integrate it using separation of variables.
	\resp La ecuación de Hamilton-Jacobi
	\begin{equation}
		\pd{S}{t} + \frac{1}{2 m} \left(\pd{S}{r}\right)^2 + \frac{1}{2 m} \left(\frac{1}{r} \pd{S}{\varphi} - \frac{ q B}{2} r\right)^2 + \frac{k}{2} r^2 = 0
	\end{equation}
	Como el hamiltoniano no depende del tiempo, el sistema es conservativo y la acción se puede escribir como
	\begin{equation}
		S = f_r(r) + f_\varphi(\varphi) - E t
	\end{equation}
	donde
	\begin{equation}
		\pd{f_\varphi}{\varphi} = p_\varphi = \mathrm{const} \implies f_\varphi = \varphi p_\varphi
	\end{equation}
	Por lo que
	\begin{align*}
		\frac{1}{2 m} \left(\pd{f_r}{r}\right)^2 + \frac{1}{2 m} \left(\frac{1}{r} \pd{f_\varphi}{\varphi} - \frac{ q B}{2} r\right)^2 + \frac{k}{2} r^2 = E\\
		\frac{1}{2 m} \left(\pd{f_r}{r}\right)^2 + \frac{1}{2 m} \left(\frac{1}{r}p_\varphi - \frac{ q B}{2} r\right)^2 + \frac{k}{2} r^2 = E
	\end{align*}
	\begin{equation*}
		\implies \left(\pd{f_r}{r}\right)^2 = 2 m E + q B p_\varphi - \left[\frac{ q^2 B^2}{4} + m k \right] r^2 - \frac{1}{r^2} p_\varphi^2
	\end{equation*}
	Resolviendo la integral
	\begin{align*}
		f_r(r) &= \int^r \sqrt{2 m E + q B p_\varphi - \left[\frac{ q^2 B^2}{4} + m k \right] r^2 - \frac{1}{r^2}} p_\varphi^2 \, dr
	\end{align*}
	Por lo que
	\begin{equation}
		S = f_r(r) + \varphi p_\varphi - E t
	\end{equation}
	\begin{equation}
		\implies \pd{S}{E} = - t_0 = \mathrm{const}, \qquad \implies \pd{S}{p_\varphi} = -\varphi_0 = \mathrm{const}
	\end{equation}
	\qed
	
\end{enumerate}

Assuming that $\Delta H = H - H_0$ is a small perturbation, and using canonical perturbation theory, calculate:
\begin{enumerate}
	\item[a.] The first nonvanishing correction to the trajectory of the particle
	\resp El potencial de este problema
	\begin{align}
		U(x) &= \frac{1}{2} k \left(\sqrt{a^2 + x^2} - b\right)^2 + \frac{1}{2} k \left(\sqrt{a^2 + x^2} - b\right)^2\nonumber\\
		&= k \left(\sqrt{a^2 + x^2} - b\right)^2
	\end{align}
	Por lo que el hamiltoniano
	\begin{equation}
		H = \frac{1}{2 m} p^2 + k \left(\sqrt{a^2 + x^2} - b\right)^2
	\end{equation}
	Expandiendo el binomio cuadrado a los 2 primeros órdenes no constantes
	\begin{equation}
		H \approx \underbrace{\frac{1}{2 m} p^2 + k \frac{a - b}{a} x^2}_{H_0} + \underbrace{k \frac{b}{4 a^3} x^4}_{\Delta H}
	\end{equation}
	Para $H_0$, las ecuaciones canónicas
	\begin{equation*}
		\dot{p}_0 = -\pd{H}{x} = - 2 k \frac{a - b}{a} x, \qquad \dot{x}_0 = \pd{H}{p} = \frac{1}{m} p
	\end{equation*}
	\begin{equation*}
		\implies \ddot{x}_0 = -\frac{2 k}{m} \frac{a - b}{a} x_0
	\end{equation*}
	por lo que la solución es un oscilador armónico 
	\begin{equation}
		x_0(t) = A \sin\left[\omega_0 t + B\right], \qquad p_0(t) = m A \omega_0 \cos\left[\omega_0 t + B\right] , \qquad \omega_0 = \sqrt{\frac{2 k}{m} \frac{a - b}{a}}
	\end{equation}
	Donde $A, B$ vienen de la integración. Utilizando \texttt{Mathematica} tenemos
	\begin{equation}
		A = \frac{\sqrt{m^2 x^2 \omega_0 ^2+p^2}}{m \omega_0 }, \qquad B = \arctan \left(\frac{p}{m x \omega_0 }\right) - \omega_0 t
	\end{equation}
	Para hacer que $A,B$ sean un par de variables $\alpha, \beta$ canónicas estas deben cumplir con $[\alpha,\beta] = 1$. En nuestro caso
	\begin{equation}
		[A,B] = \frac{1}{\sqrt{m^2 x^2 \omega_0 ^2+p^2}} = \frac{1}{m \omega_0 A}
	\end{equation}
	Para encontrar el par canónico $\alpha$ de $B$ tendremos que resolver
	\begin{equation}
		[\alpha(A),B] = \td{\alpha}{A} \cdot [A, B] = 1 \implies \td{\alpha}{A} = m \omega_0 A
	\end{equation}
	\begin{equation}
		\implies \alpha = \frac{m \omega_0 A^2}{2}
	\end{equation}
	Por lo que obtenemos un par canónico, tal que
	\begin{equation}
		\alpha = \frac{m \omega_0 A^2}{2}, \qquad \beta = B
	\end{equation}

	\begin{equation*}

	\end{equation*}
	\begin{equation}
		\implies \Delta H = \frac{b k}{a^3 m^2 \omega_0^2} \alpha^2 \sin^4\left[\omega_0 t + \beta\right]
	\end{equation}

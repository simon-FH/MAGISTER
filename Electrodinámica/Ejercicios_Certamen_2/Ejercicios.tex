		\item[(a)] Use las ecuaciones de Maxwell para demostrar que $\n{k} \cdot \n{E}_0 = \n{k} \cdot \n{B}_0 = 0$.
		\resp Tenemos que
		\begin{equation}
			\n{E}(\rv,t) = \n{E}_0 \, e^{i(\n{k}\cdot\rv - \omega t)}, \quad \n{B}(\rv,t) = \n{B}_0 \, e^{i(\n{k}\cdot\rv - \omega t)}
		\end{equation}
		entonces, para estos campos tendremos
		\begin{equation}
			\nabla \to i\n{k}, \quad \pd{}{t} \to -i\omega
		\end{equation}
		
		De las ecuaciones de Maxwell sin fuentes,
		\begin{align}
			\di\n{D} &= 0, & \di\n{B} &= 0
		\end{align}
		Reemplazando
		\begin{align}
			\di\n{D} &= 0, & \di\n{B} &= 0\nonumber\\
			i\n{k}\cdot\epsilon\n{E} &= 0, & i\n{k}\cdot\n{B} &= 0\nonumber\\
			\underbrace{i\epsilon \, e^{i(\n{k}\cdot\rv - \omega t)}}_{\neq 0} (\n{k}\cdot\n{E}_0) &= 0, & \underbrace{i \, e^{i(\n{k}\cdot\rv - \omega t)}}_{\neq 0} (\n{k}\cdot\n{B}_0) &= 0\nonumber\\
			\Rightarrow \n{k}\cdot\n{E}_0 &= 0, & \Rightarrow \n{k}\cdot\n{B}_0 &= 0
		\end{align}
		\qed
		
		% % % % % % % % % % % % % % % % % % % % % % % % % % % % % % % % % % % % % %

	\item Partiendo de
	\begin{equation}
		E_x = a_x \cos(\psi), \quad E_y = a_y \cos(\psi + \Delta)
	\end{equation}
	demuestre que la punta de $\n{E}$ satisface
	\begin{equation}
		\frac{E_x^2}{a_x^2} + \frac{E_y^2}{a_y^2} - 2\frac{E_x E_y}{a_x a_y} \cos\Delta = \sin^2 \Delta
	\end{equation}
	Identifique los casos de polarización lineal y circular.
	\resp Del primer componente
	\begin{equation*}
		\cos\psi = \frac{E_x}{a_x}
	\end{equation*}
	Del segundo
	\begin{equation*}
		\frac{E_y}{a_y} = \cos(\psi + \Delta) = \cos\psi\cos\Delta - \sin\psi\sin\Delta
	\end{equation*}
	Sustituyendo
	\begin{equation*}
		\frac{E_y}{a_y} - \frac{E_x}{a_x}\cos\Delta = -\sin\psi\sin\Delta
	\end{equation*}
	Elevando al cuadrado
	\begin{equation*}
		\left(\frac{E_y}{a_y} - \frac{E_x}{a_x}\cos\Delta\right)^2 = \sin^2\!\psi\,\sin^2\!\Delta
	\end{equation*}
	Expandiendo a la izquierda
	\begin{equation*}
		\frac{E_y^2}{a_y^2} - 2\frac{E_x E_y}{a_x a_y}\cos\Delta + \frac{E_x^2}{a_x^2}\cos^2\!\Delta = \sin^2\!\psi\,\sin^2\!\Delta
	\end{equation*}
	Reemplazando $\sin^2\psi = 1 - \cos^2\psi = 1 - E_x^2/a_x^2$ a la derecha
	\begin{equation*}
		\frac{E_y^2}{a_y^2} - 2\frac{E_x E_y}{a_x a_y}\cos\Delta + \frac{E_x^2}{a_x^2}\cos^2\!\Delta = \left(1 - \frac{E_x^2}{a_x^2}\right)\sin^2\!\Delta
	\end{equation*}
	Expandiendo a la derecha y moviendo $\frac{E_x^2}{a_x^2}\sin^2\Delta$ a la izquierda
	\begin{equation*}
		\frac{E_y^2}{a_y^2} - 2\frac{E_x E_y}{a_x a_y}\cos\Delta + \frac{E_x^2}{a_x^2}\underbrace{(\cos^2\!\Delta + \sin^2\!\Delta)}_{=\,1} = \sin^2\!\Delta
	\end{equation*}
	\begin{equation*}
		\Rightarrow \frac{E_x^2}{a_x^2} + \frac{E_y^2}{a_y^2} - 2\frac{E_x E_y}{a_x a_y}\cos\Delta = \sin^2\!\Delta
	\end{equation*}
	\qed

	\item Para $n_1 > n_2$ y $\theta_i > \theta_c$, derive el número de decaimiento
	\begin{equation}
		\kappa = k_0 \sqrt{n_1^2 \sin^2\theta_i - n_2^2}
	\end{equation}
	Explique por qué el campo transmitido no transporta potencia promedio en la dirección normal.
	\resp Los vectores de onda
	\begin{equation}
		\n{k}_i = k_0 n_1(\sin\theta_i\,\xhat + \cos\theta_i\,\zhat), \quad \n{k}_t = k_{tx}\xhat + k_{tz}\zhat
	\end{equation}
	Como el campo tangente debe ser continuo, $k_{tx} = k_{ix}$, por lo que
	\begin{equation}
		k_{tx} = k_0 n_1 \sin\theta_i
	\end{equation}
	La relación de dispersión para el componente perpendicular
	\begin{equation}
		k_{tx}^2 + k_{tz}^2 = k_0^2 n_2^2 ~~\Rightarrow~~ k_{tz}^2 = k_0^2 \left(n_2^2 - n_1^2\sin^2\theta_i\right)
	\end{equation}
	
	El ángulo crítico es $\theta_c = \arcsin(n_2/n_1)$, para $\theta_i > \theta_c$:
	\begin{equation}
		n_1\sin\theta_i > n_2 ~~\Rightarrow~~ n_2^2 - n_1^2\sin^2\theta_i < 0 ~~\Rightarrow~~ k_{tz}^2 < 0
	\end{equation}
	
	Por lo que $k_{tz}$ is imaginario. Tomando $k_{tz} = i\kappa$ con $\kappa > 0$
	\begin{equation}
		-\kappa^2 = k_0^2(n_2^2 - n_1^2\sin^2\theta_i)
	\end{equation}
	\begin{equation}
		\Rightarrow \kappa = k_0\sqrt{n_1^2\sin^2\theta_i - n_2^2}
	\end{equation}
	El campo transmitido es una onda evanescente, por lo que se propaga en el medio 2 decayendo exponencialmente.
	\begin{equation}
		\n{E}_t = E_t e^{i k_x x} e^{-\kappa z} \yhat = E_t e^{ik_0 n_1\sin\theta_i\cdot x}\,e^{-\kappa z} \yhat
	\end{equation}
	\qed
	
	Por otro lado, la potencia promedio en la dirección normal puede ser representada por
	\begin{equation}
		\pr{S_z} = \frac{1}{2}\mathrm{Re}\!\left[\n{E}_t\times\n{H}_t^*\right]_z
	\end{equation}
	A partir de $\ro\n{E} = -i\omega\uo\n{H}$ tenemos
	\begin{align*}
		\n{H}_t^* &= -\frac{i}{\omega\uo}\ro\n{E}_t\\
		&= -\frac{i}{\omega\uo} \left[\kappa\xhat + i k_x\zhat\right]|\n{E}_t|
	\end{align*}
	Luego
	\begin{equation*}
		\left[\n{E}_t\times\n{H}_t^*\right]_z = \frac{i}{\omega\uo}|\n{E}_t|^2 \kappa
	\end{equation*}
	Ya que esto es es puramente imaginario tenemos
	\begin{equation}
		\pr{S_z} = \frac{1}{2}\cancel{\mathrm{Re}\!\left[\n{E}_t\times\n{H}_t^*\right]_z} = 0
	\end{equation}
	No hay transmisión de energía normal a la superficie.
	
	\qed

	\item Suponga que la susceptibilidad de un medio satisface causalidad y que $\chi(\omega)$ decrece suficientemente rápido para $|\omega| \to \infty$. Explique porque $\chi(\omega)$ es analítica en el semiplano superior y derive formalmente la primera relación de Kramers–Kronig usando el teorema de Cauchy.
	\resp La susceptibilidad en el tiempo conecta la polarización con el campo:
	
	$$P(t) = \epsilon_0 \int_{-\infty}^{\infty} \tilde{\chi}(\tau)\, E(t-\tau)\, d\tau$$
	Causalidad exige que el medio no responda antes del estímulo
	$$\tilde{\chi}(\tau) = 0 \quad \text{para } \tau < 0$$
	Por lo tanto, la transformada de Fourier es en realidad una integral sobre el semirrecta positiva:
	$$\chi(\omega) = \int_0^{\infty} \tilde{\chi}(\tau)\, e^{i\omega\tau}\, d\tau$$
	Para $\omega = \omega' + i\omega''$ con $\omega'' > 0$:
	$$\chi(\omega) = \int_0^{\infty} \tilde{\chi}(\tau)\, e^{i\omega'\tau}\, \underbrace{e^{-\omega''\tau}}_{\text{converge}} d\tau$$
	El factor $e^{-\omega''\tau}$ garantiza convergencia absoluta para todo $\tau > 0$. La derivación bajo la integral es válida, por lo que $\chi(\omega)$ es analítica en todo el semiplano superior $\text{Im}(\omega) > 0$.
	
	\qed
	
	Considere la integral de contorno de $\dfrac{\chi(\omega')}{\omega' - \omega}$ sobre el contorno $C$ en el semiplano superior:
	
	$$\oint_C \frac{\chi(\omega')}{\omega' - \omega}\, d\omega' = 0$$
	
	ya que $\chi$ es analítica allí y el integrando no tiene polos en el interior. El contorno consiste en cuatro partes:
	$$\underbrace{\int_{-R}^{\omega-\epsilon}}_{\text{eje real izq.}} + \underbrace{\int_{\Gamma_\epsilon}}_{\text{semicírculo pequeño}} + \underbrace{\int_{\omega+\epsilon}^{R}}_{\text{eje real der.}} + \underbrace{\int_{\Gamma_R}}_{\text{semicírculo grande}} = 0$$
	Semicírculo grande $\Gamma_R$: Por hipótesis $|\chi(\omega)| \to 0$ cuando $|\omega|\to\infty$, así que:
	$$\int_{\Gamma_R} \frac{\chi(\omega')}{\omega'-\omega}\,d\omega' \xrightarrow{R\to\infty} 0$$
	Semicírculo pequeño $\Gamma_\epsilon$ (rodea $\omega$ por arriba, $\theta: \pi \to 0$): con $\omega' = \omega + \epsilon e^{i\theta}$:
	$$\int_{\Gamma_\epsilon} \frac{\chi(\omega')}{\omega'-\omega}\,d\omega' = i\int_\pi^0 \chi(\omega + \epsilon e^{i\theta})\,d\theta \xrightarrow{\epsilon\to 0} -i\pi\,\chi(\omega)$$
	Tomando $R\to\infty$, $\epsilon\to 0$, los dos tramos del eje real se convierten en el valor principal de Cauchy:
	$$\mathcal{P}\int_{-\infty}^{\infty} \frac{\chi(\omega')}{\omega' - \omega}\,d\omega' - i\pi\,\chi(\omega) = 0$$
	
	Despejando:
	
	$$\boxed{\chi(\omega) = \frac{1}{i\pi}\,\mathcal{P}\!\int_{-\infty}^{\infty} \frac{\chi(\omega')}{\omega'-\omega}\,d\omega'}$$
	
	
	Escribiendo $\chi = \chi_1 + i\chi_2$ y usando $\frac{1}{i} = -i$:
	
	$$\chi_1 + i\chi_2 = \frac{-i}{\pi}\mathcal{P}\!\int\frac{\chi_1(\omega') + i\chi_2(\omega')}{\omega'-\omega}\,d\omega' = \frac{1}{\pi}\mathcal{P}\!\int\frac{\chi_2(\omega')}{\omega'-\omega}\,d\omega' \;-\; \frac{i}{\pi}\mathcal{P}\!\int\frac{\chi_1(\omega')}{\omega'-\omega}\,d\omega'$$
	
	Igualando partes real e imaginaria:
	
	$$\boxed{\chi_1(\omega) = \frac{1}{\pi}\,\mathcal{P}\!\int_{-\infty}^{\infty} \frac{\chi_2(\omega')}{\omega'-\omega}\,d\omega'} \qquad \textbf{(primera relación K–K)}$$
	
	$$\chi_2(\omega) = -\frac{1}{\pi}\,\mathcal{P}\!\int_{-\infty}^{\infty} \frac{\chi_1(\omega')}{\omega'-\omega}\,d\omega' \qquad \text{(segunda)}$$
	
	La estructura es la de una transformada de Hilbert: la parte real y la imaginaria de $\chi$ no son independientes; conocer una determina completamente la otra. Físicamente, esto es una consecuencia directa de causalidad — la dispersión (parte real) y la absorción (parte imaginaria) están inextricablemente ligadas.

	\item Un dipolo eléctrico armónico apunta en la dirección $z$:
	\begin{equation}
		\n{p}(t) = \mathrm{Re}\!\left[p_0 e^{-i \omega t}\right]\zhat
	\end{equation}
	\begin{enumerate}
		% % % % % % % % % % % % % % % % % % % % % % % % % % % % % % % % % % % % % %
		\item[(a)] Escriba el campo eléctrico radiativo.
		\resp El campo eléctrico radiativo
		\begin{equation}
			\n{E}_\mathrm{rad} (\rv) = \ke[k^2] \frac{e^{ikr}}{r} \rhat\times(\n{p}\times\rhat)
		\end{equation}
		Tomando $\zhat\times\rhat = \sin\theta\,\phat ~\Rightarrow~ \rhat\times(\zhat\times\rhat) = -\sin\theta\,\that$. Finalmente
		\begin{equation}
			\n{E}_\mathrm{rad} (\rv) = -\ke[k^2] \frac{e^{ikr}}{r} \mathrm{Re}\!\left[p_0 e^{-i \omega t}\right] \sin\theta\,\that
		\end{equation}
		\qed

		\begin{equation}
			\td{P}{\Omega} = \frac{\uo\omega^4 |p|^2}{32\pi^2 c} \sin^2\theta

		por lo que, considerando $|p| = \left|\mathrm{Re}\!\left[p_0 e^{-i \omega t}\right]\right| = p_0$ (asumiendo que $p_0$ es real), tendremos
		\begin{equation}
			\td{P}{\Omega} = \frac{\uo\omega^4 p_0^2}{32\pi^2 c} \sin^2\theta
		\end{equation}
		\qed

		\item[(c)] Integre sobre el ángulo sólido y obtenga la potencia total.
		\resp La potencia total radiada es
		\begin{equation}
			P = \frac{\uo\omega^4 |p|^2}{12 \pi c} = \frac{\uo\omega^4 p_0^2}{12 \pi c}
		\end{equation}
		\qed
		
	\end{enumerate}

	\item Considere un medio dieléctrico lineal, homogéneo, isotrópico, no magnético y sin dispersión espacial. La respuesta del medio se describe mediante una susceptibilidad eléctrica causal $\chi(t)$, de modo que la polarización inducida por un campo eléctrico externo satisface
	\begin{equation}
		\n{P}(t) = \eo \int_{-\infty}^{t} \chi(t - t') \n{E}(t') \, dt'
	\end{equation}
	En el dominio de frecuencias
	\begin{equation}
		\n{P}(\omega) = \eo \chi(\omega)\n{E}(\omega), \qquad \epsilon(\omega) = 1 + \chi(\omega)
	\end{equation}
	La causalidad implica que $\epsilon(\omega)$ es una función analítica en el semiplano superior del plano complejo de frecuencias. Como consecuencia, sus partes real e imaginaria no son independientes, sino que están relacionadas por las relaciones de Kramers-Kronig.
	
	Para hacer el modelo más concreto, suponga que la respuesta microscópica proviene de un conjunto continuo de osciladores de Lorentz. Cada oscilador representa una carga ligada con frecuencia propia $\Omega$, amortiguamiento $\Gamma$ y desplazamiento $x_\Omega(t)$, que obedece
	\begin{equation}
		m \ddot{x}_\Omega + m \Gamma \dot{x}_\Omega + m\Omega^2 x_\Omega = - e E(t)
	\end{equation}
	Para campos armónicos con convención temporal $e^{-i \omega t}$, la contribución de un oscilador a la susceptibilidad tiene la forma
	\begin{equation}
		\chi_\Omega(\omega) = \frac{f(\Omega)}{\Omega^2 - \omega^2 - i \Gamma \omega}
	\end{equation}
	donde $f(\Omega) \, d\Omega$ es la densidad de fuerza osciladora, incluyendo la densidad de osciladores, la carga y la masa. Por tanto,
	\begin{equation}
		\chi(\omega) = \int_{0}^{\infty} \frac{f(\Omega)}{\Omega^2 - \omega^2 - i \Gamma \omega} \, d\Omega \label{(5)}
	\end{equation}
	Suponga que la densidad espectral de osciladores está concentrada en la ventana $\omega_0 < \Omega < \omega_1$ y viene dada por
	\begin{equation}
		f(\Omega) = \frac{2 A}{\pi} \Omega \, \Theta(\Omega - \omega_0) \, \Theta( \omega_1 - \Omega) \label{(6)}
	\end{equation}
	donde $A > 0, \, 0 < \omega_0 < \omega_1$ y $\Theta$ es la función escalón. En el límite de amortiguamiento débil, $\Gamma \to 0^{+}$, este modelo produce la banda rectangular de absorción
	\begin{equation}
		\mathrm{Im} \, \epsilon(\omega) = \left\{\begin{matrix}
			A, & \omega_0 < \omega < \omega_1,\\
			0, & \text{en otro caso}
		\end{matrix}\right. \label{(7)}
	\end{equation}
	para $\omega > 0 $.
	Use además la relación de Kramers-Kronig para frecuencias positivas
	\begin{equation}
		\mathrm{Re} \, \epsilon(\omega) - 1 = \frac{2}{\pi} \mathcal{P} \int_{0}^{\infty} \frac{\Omega \mathrm{Im}\, \epsilon(\Omega)}{\Omega^2 - \omega^2} \, d\Omega
	\end{equation}
	donde $\mathcal{P}$ denota el valor principal de Cauchy.
	\begin{enumerate}
		% % % % % % % % % % % % % % % % % % % % % % % % % % % % % % % % % % % % % %
		\item[1.] Derive la forma \ref{(5)} de la susceptibilidad a partir del modelo de osciladores de Lorentz y explique el significado físico de la densidad espectral $f (\Omega)$.
		\resp Sabemos que
		\begin{equation}
			\n{P} = \eo \chi(\omega) \n{E}
		\end{equation}
		Por lo que, para poder encontrar $\chi(\omega)$ tendremos que desarrollar $\n{P}$. Ya que conocemos el comportamiento de un oscilador microscópico, podemos usar esto para encontrar la polarización macroscópica integrando
		\begin{equation}
			dP = p \cdot n(\Omega) \, d\Omega
		\end{equation}
		donde $n(\Omega)$ es la densidad de osciladores por unidad de volumen $d\Omega$ y $p$ el momento dipolar de cada oscilador. Para encontrar el momento dipolar podemos usar
		\begin{equation}
			p = q \cdot x
		\end{equation}
		donde $q$ es la carga y $x$ el desplazamiento. Para encontrar el desplazamiento usaremos la ecuación
		\begin{equation*}
			m \ddot{x}_\Omega + m \Gamma \dot{x}_\Omega + m\Omega^2 x_\Omega = - e E(t)
		\end{equation*}
		Tomando el ansatz
		\begin{equation*}
			E = E_0 e^{i \omega t}, \qquad x_\Omega (t) = x_\Omega(\omega) e^{-i \omega t}
		\end{equation*}
		tal que
		\begin{equation*}
			\dot{x}_\Omega = -i\omega x_\omega, \qquad \ddot{x}_\Omega = - \omega^2 x_\omega
		\end{equation*}
		E introduciendo esto en
		\begin{align*}
			m \ddot{x}_\Omega + m \Gamma \dot{x}_\Omega + m\Omega^2 x_\Omega &= - e E(t)\\
			m \omega^2 x_\Omega - i m \Gamma \omega x_\Omega - m\Omega^2 x_\Omega &= - e E_0 e^{i \omega t}\\
			\left(m \omega^2 - i m \Gamma \omega - m\Omega^2 \right) x_\Omega &= - e E_0
		\end{align*}
		\begin{equation*}
			\implies x_\Omega(\omega) = - \frac{e E_0}{m \left(\omega^2 - i \Gamma \omega - \Omega^2\right)}
		\end{equation*}
		Luego el momento dipolar de cada oscilador de carga $-e$ será
		\begin{equation*}
			p_\Omega = -e \cdot x_\Omega (\omega) = \frac{e^2 E_0}{m \left(\omega^2 - i \Gamma \omega - \Omega^2\right)}
		\end{equation*}
		Y la polarización
		\begin{align*}
			P &= \int_{0}^{\infty} \frac{e^2 E_0}{m \left(\omega^2 - i \Gamma \omega - \Omega^2\right)} n(\Omega) \, d\Omega\\
			&= \int_{0}^{\infty} \frac{e^2 E_0 n(\Omega)}{m} \frac{1 }{\omega^2 - i \Gamma \omega - \Omega^2} \, d\Omega
		\end{align*}
		Finalmente
		\begin{align}
			\chi(\omega) &= \frac{P}{\eo E}\nonumber\\
			&= \int_{0}^{\infty} \underbrace{\frac{e^2 n(\Omega)}{m \eo}}_{f(\Omega)} \frac{1 }{\omega^2 - i \Gamma \omega - \Omega^2} \, d\Omega\nonumber\\
			&= \int_{0}^{\infty} \frac{f(\Omega)}{\omega^2 - i \Gamma \omega - \Omega^2} \, d\Omega
		\end{align}
		\qed
		
		% % % % % % % % % % % % % % % % % % % % % % % % % % % % % % % % % % % % % %
		\item[2.] Muestre que la elección \ref{(6)} produce la banda rectangular \ref{(7)} en el límite $\Gamma \to 0^{+}$.
		\resp Tenemos que
		\begin{align*}
			\epsilon(\omega) &= 1 + \chi(\omega)\\
			&= 1 + \int_{0}^{\infty} \frac{f(\Omega)}{\omega^2 - i \Gamma \omega - \Omega^2} \, d\Omega
		\end{align*}
		Si elegimos
		\begin{equation*}
			f(\Omega) = \frac{2 A}{\pi} \Omega \, \Theta(\Omega - \omega_0) \, \Theta( \omega_1 - \Omega)
		\end{equation*}
		Tendremos
		\begin{align*}
			\epsilon(\omega) &= 1 + \int_{0}^{\infty} \frac{2 A}{\pi} \Omega \, \Theta(\Omega - \omega_0) \, \Theta( \omega_1 - \Omega) \cdot \frac{1}{\omega^2 - i \Gamma \omega - \Omega^2} \, d\Omega\\
			&= 1 + \frac{2 A}{\pi} \int_{\omega_0}^{\omega_1} \frac{\Omega }{\omega^2 - i \Gamma \omega - \Omega^2} \, d\Omega\\
			&= 1 + \frac{2 A}{\pi} \int_{\omega_0}^{\omega_1} \frac{\Omega \left(\omega^2 - i \Gamma \omega - \Omega^2\right) }{\left(\omega^2 - \Omega^2\right)^2 + \left(\Gamma \omega\right)^2} \, d\Omega\\
			&= 1 + \frac{2 A}{\pi} \int_{\omega_0}^{\omega_1} \frac{\Omega \left(\omega^2 - \Omega^2\right) }{\left(\omega^2 - \Omega^2\right)^2 + \left(\Gamma \omega\right)^2} \, d\Omega - i \frac{2 A}{\pi} \int_{\omega_0}^{\omega_1} \frac{\Omega \Gamma \omega }{\left(\omega^2 - \Omega^2\right)^2 + \left(\Gamma \omega\right)^2} \, d\Omega \\
		\end{align*}
		Por lo que la parte imaginaria será
		\begin{align*}
			\mathrm{Im} \, \epsilon(\omega) &= \frac{2 A}{\pi} \int_{\omega_0}^{\omega_1} \frac{\Omega \Gamma \omega }{\left(\omega^2 - \Omega^2\right)^2 + \left(\Gamma \omega\right)^2} \, d\Omega
		\end{align*}
		Pero tenemos que, para $\Gamma \to 0^+$, la identidad
		\begin{equation}
			\lim_{\Gamma \to 0^+} \frac{\Gamma \omega }{\left(\omega^2 - \Omega^2\right)^2 + \left(\Gamma \omega\right)^2} = \lim_{\beta \to 0^+} \frac{\beta}{\alpha^2 + \beta^2} = \pi \delta\left(\alpha\right) = \pi \delta\left(\omega^2 - \Omega^2\right)
		\end{equation}
		además,
		\begin{equation}
			\delta\left(\omega^2 - \Omega^2\right) = \left(\frac{\delta(\Omega -\omega)}{2\omega} + \frac{\cancel{\delta(\Omega + \omega)}}{2\omega}\right)
		\end{equation}
		donde cancelamos el segundo término, ya que implica que $\Omega < 0$, y los límites de integración son estrictamente positivos. Entonces se dará
		\begin{align*}
			\mathrm{Im} \, \epsilon(\omega) &= \frac{2 A}{\pi} \int_{\omega_0}^{\omega_1} \Omega \cdot \pi \frac{\delta(\Omega -\omega)}{2\omega} \, d\Omega\\
			&= \frac{A}{\omega} \int_{\omega_0}^{\omega_1} \Omega \delta(\Omega -\omega) \, d\Omega\\
			&= \frac{A}{\omega} \cdot \omega\\
			&= A
		\end{align*}
		\qed
		
		% % % % % % % % % % % % % % % % % % % % % % % % % % % % % % % % % % % % % %
		\item[3.] Calcule explícitamente $\mathrm{Re} \, \epsilon(\omega)$ para $\omega > 0$. Indique con claridad dónde debe entenderse la integral en el sentido de valor principal.
		\resp Podemos usar 
		\begin{equation*}
			\mathrm{Re} \, \epsilon(\omega) - 1 = \frac{2}{\pi} \mathcal{P} \int_{0}^{\infty} \frac{\Omega \mathrm{Im}\, \epsilon(\Omega)}{\Omega^2 - \omega^2} \, d\Omega
		\end{equation*}

		\end{align*}
		Pero $\mathrm{Im}\, \epsilon(\Omega) = A$ solo en la integral de al medio, por lo que
		\begin{align*}
			\mathrm{Re} \, \epsilon(\omega) - 1 &= \frac{2}{\pi} \left[0 + \mathcal{P} \int_{\omega_0}^{\omega_1} \frac{\Omega A}{\Omega^2 - \omega^2} \, d\Omega + 0\right]\\
			&= \frac{A}{\pi} \ln\left|\frac{\omega_1^2 - \omega^2}{\omega_0^2 - \omega^2}\right|
		\end{align*}
		
		
		% % % % % % % % % % % % % % % % % % % % % % % % % % % % % % % % % % % % % %
		\item[4.] Dibuje esquemáticamente $\mathrm{Im} \, \epsilon(\omega)$ y $\mathrm{Re} \epsilon(\omega)$ como funciones de $\omega$. Señale el comportamiento cerca de $\omega_0$, cerca de $\omega_1$, en $\omega = 0$ y cuando $\omega \to \infty$.
		\resp

		\item[5.] Discuta físicamente por qué una banda de absorción con bordes abruptos produce singularidades logarítmicas en la parte real de la respuesta. ¿Qué cambiaría cualitativamente si los bordes de la banda fueran suaves?
		\resp
		
		
	\end{enumerate}
